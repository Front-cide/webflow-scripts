<script>
/* RASTER — site script (GSAP only). Blocks G1 to G34; each block runs only where its elements exist (the About 3 team spotlight and some Home lists live in those pages' own scripts).
   GSAP is loaded by Webflow's built-in GSAP integration (Site settings → GSAP) with ScrollTrigger and SplitText
   switched on; DrawSVG and CustomEase are used when they are switched on too. Nothing loads from a CDN.
   Native scroll only: sticky parts use CSS position: sticky on their classes.
   Elements are found by their data-* attributes. Remove an attribute in the Designer to switch that effect off.
   Page scripts (Page settings → Custom code → Before </body>) add the motion that belongs to one page only. */
(function () {
  if (!window.gsap) return;

  /* KEY VARIABLES */
  const RS = window.RS = {
    ease: 'expo.out',             // main easing for entrances
    easeInOut: 'power3.inOut',    // wipes, menu, accordions (replaced by a softer CustomEase when that plugin is on)
    speed: 1,                     // global duration multiplier (0.8 = faster, 1.2 = slower)
    navHide: true,                // hide the navbar when scrolling down, show it when scrolling up
    magnetic: 0.25,               // pull of [data-magnetic] buttons (0 = off)
    cursorLag: 0.45,              // seconds the cursor square takes to catch up
    rollStagger: 0.02,            // letter roll on hover, seconds between letters
    clockZone: 'Europe/Zurich',   // time zone of every [data-clock]
    autoplay: 3.6,                // seconds per slide for the team spotlight (About 3)
    quoteDelay: 8,                // seconds per quote (Home 2 voices)
    light: {                      // colours of the day theme set by the night/day switch (Variables → Theme → Light)
      '--_theme---color--page': '#ffffff', '--_theme---color--soft': '#f2f2f0', '--_theme---color--panel': '#f2f2f0',
      '--_theme---color--text': '#0b0b0b', '--_theme---color--muted': '#646460', '--_theme---color--muted-strong': '#5f5f5c',
      '--_theme---color--border-strong': '#cfcfcb', '--_theme---color--line': 'rgba(11, 11, 11, 0.1)', '--_theme---color--guide': 'rgba(11, 11, 11, 0.07)',
      '--_theme---color--mark': 'rgba(11, 11, 11, 0.6)', '--_theme---color--transition': '#e8e8e5'
    },
    transition: true,             // page transition between pages (false = plain page loads)
    hoverIn: 0.6, hoverOut: 0.35, // hover in / hover out, seconds (out is shorter so the pointer never waits)
    static: location.search.indexOf('static') > -1   // ?static=1 shows the final state with no motion
  };

  // A plugin switched off in Site settings → GSAP gives a static page instead of a broken one.
  const READY = !!(window.gsap && window.ScrollTrigger && window.SplitText);
  if (READY) gsap.registerPlugin(ScrollTrigger, SplitText);
  if (window.DrawSVGPlugin) gsap.registerPlugin(DrawSVGPlugin);
  if (window.CustomEase) { gsap.registerPlugin(CustomEase); CustomEase.create('rsOut', 'M0,0 C0.16,1 0.3,1 1,1'); CustomEase.create('rsInOut', 'M0,0 C0.7,0 0.15,1 1,1'); RS.easeInOut = 'rsInOut'; }
  const html = document.documentElement;
  const MOTION = READY && !RS.static && !matchMedia('(prefers-reduced-motion: reduce)').matches;
  const FINE = matchMedia('(hover: hover) and (pointer: fine)').matches;
  const D = s => (MOTION ? s * RS.speed : 0);
  RS.motion = MOTION; RS.fine = FINE; RS.ready = READY; RS.D = D; RS.mm = gsap.matchMedia();
  gsap.config({ nullTargetWarn: false });
  gsap.defaults({ ease: window.CustomEase ? 'rsOut' : RS.ease, duration: D(1.1) });
  if (READY) ScrollTrigger.config({ ignoreMobileResize: true });
  // Webflow renders script buttons as links: keep them from jumping to the top of the page
  addEventListener('click', e => { const a = e.target.closest && e.target.closest('a[type="button"], a[href="#"]'); if (a) e.preventDefault(); }, true);

  /* Shared helpers (page scripts use them through window.RS) */
  RS.color = (n, f) => String(gsap.getProperty(html, '--_theme---color--' + n) || f).trim();
  RS.look = (on, off, props) => { const o = { on: {}, off: {} }; props.forEach(p => { o.on[p] = gsap.getProperty(on, p); o.off[p] = gsap.getProperty(off, p); }); return o; };
  // line masks clip descenders (g, y, p): give each mask a little room, then pull it back
  RS.maskPad = s => { if (s && s.masks && s.masks.length) gsap.set(s.masks, { paddingTop: '0.06em', paddingBottom: '0.22em', marginTop: '-0.06em', marginBottom: '-0.22em' }); return s; };
  RS.lines = (el, start) => SplitText.create(el, { type: 'lines', mask: 'lines', autoSplit: true, onSplit: s => RS.maskPad(s) && gsap.from(s.lines, { yPercent: 130, duration: D(1.2), stagger: 0.08, ease: RS.ease, scrollTrigger: { trigger: el, start: start || 'top 88%', once: true } }) });
  RS.rise = (els, trigger, o) => gsap.from(els, { y: 30, autoAlpha: 0, duration: D(1.1), stagger: 0.07, ease: RS.ease, clearProps: 'transform', scrollTrigger: { trigger: trigger, start: 'top 85%' }, ...(o || {}) });
  RS.wipe = (frame, trigger, o) => gsap.timeline({ scrollTrigger: { trigger: trigger || frame, start: 'top 82%' } })
    .fromTo(frame, { clipPath: 'inset(100% 0% 0% 0%)' }, { clipPath: 'inset(0% 0% 0% 0%)', duration: D(1.4), ease: RS.easeInOut, ...(o || {}) }, 0)
    .fromTo(gsap.utils.toArray('img', frame), { scale: 1.2 }, { scale: 1, duration: D(1.8), ease: 'power3.out', clearProps: 'transform' }, 0.45);
  // text swap: the old text lifts out, the new one rises in (used for counters, prices and labels)
  RS.swap = (el, text, dir, delay) => gsap.timeline({ delay: delay || 0 })
    .to(el, { yPercent: -60 * (dir || 1), autoAlpha: 0, duration: D(0.3), ease: 'power2.in' })
    .set(el, { textContent: text, modifiers: { textContent: () => text }, yPercent: 60 * (dir || 1) })
    .to(el, { yPercent: 0, autoAlpha: 1, duration: D(0.7), ease: RS.ease });
  RS.pad = n => (n < 10 ? '0' : '') + n;
  // writes text through GSAP; keeps strings like "09" as they are
  RS.text = (el, s) => gsap.set(el, { textContent: s, modifiers: { textContent: () => s } });
  // DATA: hover map (H) and day colours (D) live in #rs-data (Head code). P = properties, V = values.
  const DATA = (() => { const el = gsap.utils.toArray('#rs-data')[0]; try { return el ? JSON.parse(el.textContent) : null; } catch (err) { return null; } })();
  const camel = k => k.replace(/-([a-z])/g, (m, c) => c.toUpperCase());
  RS.vars = list => { const o = {}; list.forEach((x, k) => { if (k % 2 === 0) o[camel(DATA.P[x])] = DATA.V[list[k + 1]]; }); return o; };
  // sticky stages: above 991px a [data-pin-track] wrapper gets the stage height plus its scroll, so the stage holds (CSS sticky)
  RS.track = (stage, extra) => { const t = stage.parentElement, size = () => gsap.set(t, { height: innerWidth > 991 ? stage.offsetHeight + (extra ? extra() : parseFloat(t.getAttribute('data-pin-track')) * innerHeight) : 'auto' }); size(); ScrollTrigger.addEventListener('refreshInit', size); return t; };

  /* G1 — PAGE READY: once the page and its own script are in, reveal the page hidden by the head script and play the intro.
     A page script sets RS.hold = true when it plays the intro itself (Home preloader). */
  RS.intro = gsap.timeline({ paused: true });
  RS.hold = false;
  // waits for DOMContentLoaded, so every page script has added its part first
  const go = () => gsap.delayedCall(0, () => {
    if (window.__rasterReveal) window.__rasterReveal(); else gsap.set(html, { visibility: 'visible' });
    if (!RS.hold) RS.intro.play(0);
    if (READY) gsap.delayedCall(1, () => ScrollTrigger.refresh());
  });
  if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded', go); else go();
  if (!READY) return;

  /* G2 — GRID GUIDES + NAVBAR INTRO: the column lines grow from the middle, the nav items drop in */
  if (MOTION) {
    RS.intro.from(gsap.utils.toArray('.guides > div'), { scaleY: 0, duration: 1.6, stagger: { each: 0.05, from: 'center' }, ease: RS.easeInOut }, 0)
      .from(gsap.utils.toArray('.nav_links_link, .theme_t, .nav_time, .nav_cta'), { yPercent: -100, autoAlpha: 0, duration: 0.9, stagger: 0.05, ease: RS.ease }, 0.2);
  }

  /* G3 — NAVBAR: slides up when scrolling down, back when scrolling up (paused while the menu is open) */
  const nav = gsap.utils.toArray('[data-nav]')[0];
  let menuOpen = false;
  const navHide = nav && RS.navHide && MOTION ? gsap.to(nav, { yPercent: -100, duration: 0.6, ease: RS.easeInOut, paused: true }) : null;
  if (navHide) ScrollTrigger.create({ start: 'top -80%', end: 'max', onUpdate: s => { if (!menuOpen) (s.direction === 1 ? navHide.play() : navHide.reverse()); }, onLeaveBack: () => navHide.reverse() });

  /* G4 — MENU (tablet and phone): full-screen panel wipes down, links rise, burger turns into a cross. Escape closes it. */
  const burger = gsap.utils.toArray('[data-menu-toggle]')[0], menu = gsap.utils.toArray('[data-menu]')[0];
  let closeMenu = () => {};
  if (burger && menu) {
    const bars = gsap.utils.toArray('.nav_burger_icon', burger), links = gsap.utils.toArray('a', menu);
    const tl = gsap.timeline({ paused: true, defaults: { ease: RS.ease } })
      .set(menu, { visibility: 'visible' })
      .fromTo(menu, { clipPath: 'inset(0% 0% 100% 0%)' }, { clipPath: 'inset(0% 0% 0% 0%)', duration: D(0.8), ease: RS.easeInOut })
      .from(links, { yPercent: 60, autoAlpha: 0, stagger: 0.06, duration: D(0.9) }, D(0.35))
      .from(gsap.utils.toArray('.menu_foot', menu), { autoAlpha: 0, duration: D(0.6) }, D(0.6))
      .to(bars[0] || {}, { y: 3, rotation: 45, duration: D(0.5), ease: RS.easeInOut }, 0)
      .to(bars[1] || {}, { y: -3, rotation: -45, duration: D(0.5), ease: RS.easeInOut }, 0);
    const setMenu = on => {
      menuOpen = on;
      gsap.set(burger, { attr: { 'aria-expanded': on ? 'true' : 'false', 'aria-label': on ? 'Close menu' : 'Open menu' } });
      gsap.set(menu, { attr: { 'aria-hidden': on ? 'false' : 'true' } });
      gsap.set(document.body, { overflow: on ? 'hidden' : '' });
      if (navHide) navHide.reverse();
      if (on) tl.timeScale(1).play(); else tl.timeScale(1.6).reverse();
    };
    closeMenu = () => setMenu(false);
    burger.addEventListener('click', () => setMenu(!menuOpen));
    links.forEach(a => a.addEventListener('click', () => { if (menuOpen) closeMenu(); }));
  }
  addEventListener('keydown', e => { if (e.key === 'Escape' && menuOpen) closeMenu(); });

  /* G5 — CLOCK + PULSE: every [data-clock] shows the studio time; status dots breathe */
  const clocks = gsap.utils.toArray('[data-clock]');
  if (clocks.length) {
    const fmt = new Intl.DateTimeFormat('en-GB', { timeZone: RS.clockZone, hour: '2-digit', minute: '2-digit', hour12: false });
    const tick = () => gsap.set(clocks, { textContent: fmt.format(new Date()) });
    tick(); gsap.to({}, { duration: 10, repeat: -1, onRepeat: tick });
  }
  if (MOTION) gsap.to(gsap.utils.toArray('[class^="pulse"]'), { opacity: 0.25, duration: 0.8, repeat: -1, yoyo: true, ease: 'sine.inOut' });

  /* G6 — LETTER ROLL [data-roll]: on hover each letter rolls up and comes back from below */
  if (MOTION && FINE) gsap.utils.toArray('[data-roll]').forEach(el => {
    const disp = getComputedStyle(el).display;
    if (disp === 'none') return; // hidden at this breakpoint (e.g. the navbar CTA on mobile)
    gsap.set(el, { display: disp === 'flex' ? 'block' : 'inline-block', whiteSpace: 'pre' }); // keep word spaces (a flex parent would drop them)
    const chars = SplitText.create(el, { type: 'chars', mask: 'chars' }).chars, host = el.closest('a, button') || el;
    let run = null;
    host.addEventListener('pointerenter', () => {
      if (run && run.isActive()) return;
      run = gsap.to(chars, { keyframes: [{ yPercent: -110, duration: 0.25, ease: 'power2.in' }, { yPercent: 110, duration: 0 }, { yPercent: 0, duration: 0.4, ease: 'power3.out' }], stagger: RS.rollStagger });
    });
  });

  /* G7 — CURSOR [data-cursor-root]: a square follows the pointer; elements with data-cursor="Label" show that label */
  const cursor = gsap.utils.toArray('[data-cursor-root]')[0];
  if (cursor && FINE && MOTION) {
    const label = gsap.utils.toArray('[data-cursor-label]', cursor)[0];
    const xTo = gsap.quickTo(cursor, 'x', { duration: RS.cursorLag, ease: 'power3' }), yTo = gsap.quickTo(cursor, 'y', { duration: RS.cursorLag, ease: 'power3' });
    let current = null;
    addEventListener('pointermove', e => { xTo(e.clientX); yTo(e.clientY); gsap.to(cursor, { opacity: 1, duration: 0.3, overwrite: 'auto' }); });
    document.addEventListener('pointerleave', () => gsap.to(cursor, { opacity: 0, duration: 0.3 }));
    document.addEventListener('pointerover', e => {
      const t = e.target.closest('[data-cursor]');
      if (t === current) return; current = t;
      if (t && label) gsap.set(label, { textContent: t.getAttribute('data-cursor') });
      if (label) gsap.to(label, { clipPath: t ? 'inset(0% 0% 0% 0%)' : 'inset(0% 100% 0% 0%)', duration: 0.5, ease: RS.ease });
      gsap.set(cursor, { mixBlendMode: t ? 'normal' : '' }); // label at its own size and colours
    });
  }

  /* G8 — MAGNETIC BUTTONS [data-magnetic]: the button leans toward the pointer and springs back */
  if (FINE && MOTION && RS.magnetic) gsap.utils.toArray('[data-magnetic]').forEach(el => {
    const xTo = gsap.quickTo(el, 'x', { duration: 0.8, ease: 'elastic.out(1, 0.45)' }), yTo = gsap.quickTo(el, 'y', { duration: 0.8, ease: 'elastic.out(1, 0.45)' });
    el.addEventListener('pointermove', e => { const r = el.getBoundingClientRect(); xTo((e.clientX - r.left - r.width / 2) * RS.magnetic); yTo((e.clientY - r.top - r.height / 2) * RS.magnetic * 1.5); });
    el.addEventListener('pointerleave', () => { xTo(0); yTo(0); });
  });

  /* G9 — BARCODE [data-barcode="WORD"]: a real Code 128 of the word; it shuffles through other codes on hover */
  gsap.utils.toArray('[data-barcode]').forEach(svg => {
    const T = ('212222 222122 222221 121223 121322 131222 122213 122312 132212 221213 221312 231212 112232 122132 122231 113222 123122 123221 223211 221132 ' +
      '221231 213212 223112 312131 311222 321122 321221 312212 322112 322211 212123 212321 232121 111323 131123 131321 112313 132113 132311 211313 ' +
      '231113 231311 112133 112331 132131 113123 113321 133121 313121 211331 231131 213113 213311 213131 311123 311321 331121 312113 312311 332111 ' +
      '314111 221411 431111 111224 111422 121124 121421 141122 141221 112214 112412 122114 122411 142112 142211 241211 221114 413111 241112 134111 ' +
      '111242 121142 121241 114212 124112 124211 411212 421112 421211 212141 214121 412121 111143 111341 131141 114113 114311 411113 411311 113141 ' +
      '114131 311141 411131 211412 211214 211232').split(' ');
    const word = Array.from(svg.getAttribute('data-barcode')).map(c => c.charCodeAt(0) - 32);
    const encode = v => [T[104]].concat(v.map(n => T[n]), [T[v.reduce((a, n, i) => a + n * (i + 1), 104) % 103], '2331112']).join('');
    const draw = v => {
      let x = 0, out = '';
      Array.from(encode(v)).forEach((w, i) => { if (i % 2 === 0) out += '<rect x="' + x + '" y="0" width="' + w + '" height="40"></rect>'; x += +w; });
      gsap.set(svg, { innerHTML: out, attr: { viewBox: '0 0 ' + x + ' 40', preserveAspectRatio: 'none' } });
    };
    draw(word);
    if (MOTION) RS.intro.from(gsap.utils.toArray('rect', svg), { scaleY: 0, transformOrigin: '50% 100%', duration: 0.8, stagger: { each: 0.012, from: 'random' }, ease: RS.ease }, 0.5);
    let busy = null;
    if (MOTION && FINE) svg.parentElement.addEventListener('pointerenter', () => {
      if (busy && busy.isActive()) return;
      busy = gsap.to({}, { duration: 0.055, repeat: 7, onRepeat: () => draw(word.map(() => gsap.utils.random(1, 94, 1))), onComplete: () => draw(word) });
    });
  });

  /* G10 — NIGHT / DAY SWITCH [data-theme-toggle]: the knob slides, the theme colours (RS.light) fade in and the day colours
     of the data block (D) are laid over the parts that keep a fixed colour at night. The choice is remembered on the next page. */
  const themeBtn = gsap.utils.toArray('[data-theme-toggle]')[0];
  RS.day = false;
  const dayGroups = DATA ? DATA.D.map(g => ({ els: gsap.utils.toArray(g[1].map(c => '.' + c).join(',')), v: RS.vars(g[0]) })).filter(g => g.els.length) : [];
  if (themeBtn) {
    const knob = gsap.utils.toArray('.tt_knob', themeBtn)[0], keys = Object.keys(RS.light), dark = {};
    keys.forEach(k => { dark[k] = String(gsap.getProperty(html, k)).trim(); });
    let day = false;
    try { day = localStorage.getItem('raster-theme') === 'light'; } catch (err) { day = false; }
    const apply = (on, d) => {
      day = on; RS.day = on;
      if (RS.unhover) RS.unhover();
      /* night values are read once, before anything moves; a colour a child only inherits stays inherited, so hovers on the parent still reach it */
      dayGroups.forEach(g => { if (on && !g.n) g.n = g.els.map(el => { const o = {}, p = el.parentElement; Object.keys(g.v).forEach(k => { o[k] = gsap.getProperty(el, k); }); if ('color' in o && p && gsap.getProperty(p, 'color') === o.color) delete o.color; return o; }); });
      const vars = {}; keys.forEach(k => { vars[k] = on ? RS.light[k] : dark[k]; });
      gsap.to(html, { ...vars, duration: d, ease: 'power2.inOut', attr: { 'data-mode': on ? 'day' : 'night' } });
      dayGroups.forEach(g => {
        if (!g.n) return;
        g.els.forEach((el, i) => { const ks = Object.keys(g.n[i]), to = {}; ks.forEach(k => { to[k] = on ? g.v[k] : g.n[i][k]; }); gsap.to(el, { ...to, duration: d, ease: 'power2.inOut', overwrite: 'auto', onComplete: () => { if (!on) gsap.set(el, { clearProps: ks.join(',') }); } }); });
      });
      if (knob) gsap.to(knob, { x: on ? knob.offsetWidth : 0, duration: d, ease: RS.ease });
      gsap.set(themeBtn, { attr: { 'aria-pressed': on ? 'true' : 'false' } });
    };
    apply(day, 0);
    themeBtn.addEventListener('click', () => { apply(!day, D(0.7)); try { localStorage.setItem('raster-theme', day ? 'light' : 'dark'); } catch (err) { day = !!day; } });
  }

  /* G11 — HAIRLINES + SECTION TITLES: every divider draws from the left, h2 titles rise line by line, labels slide in */
  if (MOTION) {
    gsap.utils.toArray('[class^="divider"]:not([class*="_after"])').filter(d => !d.closest('[data-hero], [data-v2hero], [data-pg-hero], [data-h3hero]')).forEach(d =>
      gsap.fromTo(d, { clipPath: 'inset(-4px 100% -4px -4px)' }, { clipPath: 'inset(-4px -4px -4px -4px)', duration: 1.6, ease: RS.easeInOut, scrollTrigger: { trigger: d, start: 'top 95%' } }));
    gsap.utils.toArray('h2[class^="h2"]').forEach(el => RS.lines(el));
    gsap.utils.toArray('[class^="shead"] > p:first-child, .faq_left > div:first-child, .deck_left > p:first-child').forEach(l =>
      gsap.from(l, { autoAlpha: 0, x: -12, duration: 0.9, ease: RS.ease, scrollTrigger: { trigger: l, start: 'top 92%' } }));
    gsap.utils.toArray('[data-words]').forEach(st => {
      const words = SplitText.create(st, { type: 'words' }).words;
      gsap.fromTo(words, { opacity: 0.15 }, { opacity: 1, stagger: 0.1, ease: 'none', scrollTrigger: { trigger: st, start: 'top 80%', end: 'bottom 50%', scrub: true } });
    });
  }

  /* G12 — COUNT-UP: every number with a count attribute counts from zero when it scrolls in (keeps leading zeros and suffixes) */
  ['data-pg-count', 'data-odo', 'data-v2count', 'data-h3count', 'data-h3bc', 'data-h3tc', 'data-h3n'].forEach(a => gsap.utils.toArray('[' + a + ']').forEach(el => {
    const v = el.getAttribute(a), t = el.textContent.trim(), to = parseFloat(v), dec = v.indexOf('.') > -1, pad = t.length > 1 && t[0] === '0';
    const suf = el.getAttribute('data-suffix') || '', o = { n: 0 };
    const fmt = n => (dec ? n.toFixed(1) : pad ? RS.pad(Math.round(n)) : Math.round(n)) + suf;
    if (!MOTION || isNaN(to)) return;
    gsap.set(el, { fontVariantNumeric: 'tabular-nums' }); RS.text(el, fmt(0));
    gsap.to(o, { n: to, duration: 1.8, ease: RS.ease, onUpdate: () => RS.text(el, String(fmt(o.n))), scrollTrigger: { trigger: el, start: 'top 92%', once: true } });
  }));
  // thin meters [data-meter="94"] fill to their value
  gsap.utils.toArray('[data-meter]').forEach(m => gsap.fromTo(gsap.utils.toArray(':scope > div', m), { scaleX: 0 }, { scaleX: (+m.getAttribute('data-meter') || 100) / 100, transformOrigin: '0% 50%', duration: D(1.8), ease: RS.easeInOut, scrollTrigger: { trigger: m, start: 'top 90%' } }));

  /* G13 — INNER PAGE REVEALS: rise, lines, image wipe, parallax, cover, filling steps */
  if (MOTION) {
    gsap.utils.toArray('[data-pg-rise]').forEach(r => RS.rise(gsap.utils.toArray(r.children), r));
    gsap.utils.toArray('[data-pg-lines]').forEach(el => RS.lines(el, 'top 85%'));
    gsap.utils.toArray('[data-pg-reveal]').forEach(f => RS.wipe(f, f, { ease: RS.easeInOut }));
    gsap.utils.toArray('[data-pg-par]').forEach(im => gsap.fromTo(im, { yPercent: -7 }, { yPercent: 0, ease: 'none', scrollTrigger: { trigger: im.parentElement, start: 'top bottom', end: 'bottom top', scrub: 1 } }));
    gsap.utils.toArray('[data-pg-cover]').forEach(c => gsap.fromTo(c.firstElementChild, { clipPath: 'inset(0% 18% 0% 18%)' }, { clipPath: 'inset(0% 0% 0% 0%)', ease: 'none', scrollTrigger: { trigger: c, start: 'top bottom', end: 'top 20%', scrub: 1 } }));
    gsap.utils.toArray('[data-pg-steps]').forEach(s => gsap.timeline({ scrollTrigger: { trigger: s, start: 'top 75%', end: 'bottom 45%', scrub: 1 } })
      .fromTo(gsap.utils.toArray('[class^="cr_st_l"] > div', s), { scaleX: 0 }, { scaleX: 1, transformOrigin: '0% 50%', stagger: 1, ease: 'none' }, 0)
      .to(gsap.utils.toArray('.cr_st_n', s), { color: () => RS.color('text', '#f2f2f0'), stagger: 1, ease: 'none' }, 0));
  }

  /* G14 — PAGE HEADER INTRO [data-pg-hero]: the title rises line by line, the top row and the side follow, the rule draws */
  const pgHero = gsap.utils.toArray('[data-pg-hero]')[0];
  if (pgHero && MOTION) {
    const pt = gsap.utils.toArray('[data-pg-title]', pgHero)[0];

      const t = HOV.get(h);
      if (t) t.o.forEach(e => gsap.set(e[0], { transformOrigin: e[1] }));
      if (t) gsap.to(t, { progress: 0, duration: RS.hoverOut, ease: 'power3.out', overwrite: true, onComplete: () => { HOV.delete(h); t.kill(); t.data.forEach(u => gsap.set(u[0], { clearProps: u[1] })); } });
    });
  }));
  // Home 2 people [data-v2team]: the first name is lit until the pointer reaches the list
  gsap.utils.toArray('[data-v2team]').forEach(list => {
    const b = gsap.utils.toArray('.v2tm_b', list)[0], e = DATA && DATA.H.v2tm && DATA.H.v2tm[0].filter(x => x[0] === 'v2tm_b')[0];
    if (!b || !e) return;
    gsap.set(b, RS.vars(e.slice(1)));
    list.addEventListener('pointerenter', () => gsap.set(b, { clearProps: 'color,translate' }), { once: true }); // the CSS transition fades it back
  });
  // [list, row, part that fades ('' = the row), opacity]: the row under the pointer stays at full strength, the others dim
  if (DATA && DATA.G && FINE && MOTION) DATA.G.forEach(g => gsap.utils.toArray(g[0]).forEach(list => {
    const rows = gsap.utils.toArray(g[1], list), part = r => (g[2] ? gsap.utils.toArray(g[2], r) : [r]);
    const show = on => rows.forEach(r => gsap.to(part(r), { opacity: !on || r === on || r.classList.contains('is-open') ? 1 : g[3], duration: on ? RS.hoverIn : RS.hoverOut, ease: on ? RS.ease : 'power3.out', overwrite: 'auto' }));
    rows.forEach(r => r.addEventListener('pointerenter', () => show(r)));
    list.addEventListener('pointerleave', () => show(null));
  }));

  /* G33 — PAGE TRANSITION [data-ptr] (component Page Transition): following a link to another page, four column panels rise
     with the page name; on the new page the name lifts away and the panels fold up. Home's own preloader takes over there.
     RS.transition = false turns it off. A page restored by the back button is never left covered. */
  const ptr = gsap.utils.toArray('[data-ptr]')[0];
  if (ptr && RS.transition) {
    const cols = gsap.utils.toArray('.ptr_col', ptr), nameEl = gsap.utils.toArray('[data-ptr-name]', ptr)[0], NAMES = { 'join-us': 'Careers' };
    // Overview and the Home pages have the logo preloader: the panels close on them with no name
    const label = path => { const s = path.split('/')[1] || ''; return /^(home-\d)?$/.test(s) ? '' : NAMES[s] || s.replace(/-\d+$/, '').replace(/^\w/, c => c.toUpperCase()); };
    let split = null, came = false;
    const letters = text => { if (split) split.revert(); RS.text(nameEl, text); split = SplitText.create(nameEl, { type: 'chars' }); return split.chars; };
    try { came = sessionStorage.getItem('raster-ptr') === '1'; sessionStorage.removeItem('raster-ptr'); } catch (err) { came = false; }
    gsap.set(ptr, { visibility: 'hidden' });
    const here = label(location.pathname);
    if (came && MOTION && here) { // Overview and the Home pages play their preloader instead (the loader markup is on every page)
      const ch = letters(here);
      gsap.set(ptr, { visibility: 'visible' }); gsap.set(cols, { scaleY: 1 });
      gsap.timeline({ onComplete: () => gsap.set(ptr, { visibility: 'hidden' }) })
        .to(ch, { yPercent: -60, autoAlpha: 0, filter: 'blur(14px)', duration: 0.5, stagger: 0.035, ease: 'power3.in' }, 0)
        .to(cols, { scaleY: 0, transformOrigin: '50% 0%', duration: 0.75, ease: 'expo.inOut', stagger: 0.05 }, 0.22)
        .from(gsap.utils.toArray('main'), { y: () => innerHeight * 0.2, duration: 1.2, ease: 'expo.out', clearProps: 'transform' }, 0.3);
    }
    addEventListener('pageshow', e => { if (e.persisted) { gsap.set(ptr, { visibility: 'hidden' }); gsap.set(cols, { scaleY: 0 }); } });
    if (MOTION) document.addEventListener('click', e => {
      const a = e.target.closest('a[href]');
      if (!a || e.defaultPrevented || e.button || e.metaKey || e.ctrlKey || e.shiftKey || e.altKey || a.target === '_blank' || a.hasAttribute('download') || a.origin !== location.origin || a.pathname === location.pathname) return;
      e.preventDefault();
      const ch = letters(label(a.pathname));
      try { sessionStorage.setItem('raster-ptr', '1'); } catch (err) { came = false; }
      gsap.set(ptr, { visibility: 'visible' });
      gsap.timeline({ onComplete: () => { location.href = a.href; } })
        .fromTo(cols, { scaleY: 0 }, { scaleY: 1, transformOrigin: '50% 100%', duration: 0.65, ease: 'expo.inOut', stagger: 0.05 }, 0)
        .fromTo(ch, { yPercent: 70, autoAlpha: 0, filter: 'blur(16px)' }, { yPercent: 0, autoAlpha: 1, filter: 'blur(0px)', duration: 0.75, stagger: 0.04, ease: 'expo.out' }, 0.32)
        .to({}, { duration: 0.1 });
    });
  }

  /* G34 — BOTTOM BLUR [data-edgeblur] (inside the Footer component) + VIDEOS: a frosted strip with a soft fade along the bottom
     of the screen, on every page; the fade uses the page colour, so it follows the night/day switch */
  const eb = gsap.utils.toArray('[data-edgeblur]')[0];
  if (eb) { document.body.appendChild(eb); /* out of the footer: its overflow clip would stop the blur */
    gsap.set(eb, { attr: { style: 'opacity:1;visibility:visible;background-image:linear-gradient(to bottom,transparent,color-mix(in srgb,var(--_theme---color--page) 70%,transparent))' } });
    // eight layers, each starting a little lower and blurring a little more (0.3px to 20px)
    gsap.utils.toArray(':scope > div', eb).forEach((l, i) => { const s = i * 8.75, b = [0.3, 1.3, 2.8, 5, 7.8, 11.3, 15.3, 20][i] + 'px', m = 'linear-gradient(to bottom, transparent ' + s + '%, #000 ' + (s + 8.75) + '%, #000 100%)', f = 'blur(' + b + ') saturate(1.2)'; gsap.set(l, { attr: { style: '-webkit-backdrop-filter:' + f + ';backdrop-filter:' + f + ';-webkit-mask-image:' + m + ';mask-image:' + m } }); }); }
  // Webflow drops the muted and loop attributes of video tags, and browsers only autoplay muted video: set them here
  gsap.utils.toArray('video').forEach(v => { v.muted = true; v.loop = true; v.playsInline = true; if (v.hasAttribute('autoplay')) v.play().catch(() => {}); });
  gsap.utils.toArray('video[data-inview]:not([autoplay])').forEach(v => ScrollTrigger.create({ trigger: v, start: 'top bottom', end: 'bottom top', onToggle: st => (st.isActive ? v.play().catch(() => {}) : v.pause()) }));

  /* G35 — SHARP IMAGES: load files as wide as shown */
  gsap.utils.toArray('img[srcset]').forEach(i => { const w = i.offsetWidth; if (w > parseInt(i.sizes)) gsap.set(i, { attr: { sizes: w + 'px' } }); });

  /* G31 — REFRESH: once every image has arrived, recalculate the scroll positions */
  addEventListener('load', () => ScrollTrigger.refresh());
})();
